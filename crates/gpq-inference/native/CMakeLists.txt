function(gpq_collect_link_items target)
    if(NOT TARGET "${target}")
        if(NOT "${target}" STREQUAL "")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "SYSTEM|${target}")
        endif()
        return()
    endif()

    get_target_property(alias_target "${target}" ALIASED_TARGET)
    if(alias_target)
        set(target "${alias_target}")
    endif()

    get_property(stack GLOBAL PROPERTY GPQ_LINK_STACK)
    if("${target}" IN_LIST stack)
        return()
    endif()
    set_property(GLOBAL APPEND PROPERTY GPQ_LINK_STACK "${target}")

    get_target_property(target_type "${target}" TYPE)
    get_target_property(imported "${target}" IMPORTED)
    if(target_type STREQUAL "STATIC_LIBRARY")
        set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "STATIC|$<TARGET_LINKER_FILE:${target}>")
    elseif(target_type STREQUAL "SHARED_LIBRARY" OR target_type STREQUAL "MODULE_LIBRARY")
        set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "DYNAMIC|$<TARGET_LINKER_FILE:${target}>")
    elseif(target_type STREQUAL "UNKNOWN_LIBRARY" AND imported)
        get_target_property(imported_location "${target}" IMPORTED_LOCATION)
        get_target_property(imported_implib "${target}" IMPORTED_IMPLIB)
        get_target_property(imported_configs "${target}" IMPORTED_CONFIGURATIONS)
        foreach(imported_config IN LISTS imported_configs)
            string(TOUPPER "${imported_config}" imported_config_upper)
            if(NOT imported_location OR imported_location MATCHES "-NOTFOUND$")
                get_target_property(imported_location "${target}" "IMPORTED_LOCATION_${imported_config_upper}")
            endif()
            if(NOT imported_implib OR imported_implib MATCHES "-NOTFOUND$")
                get_target_property(imported_implib "${target}" "IMPORTED_IMPLIB_${imported_config_upper}")
            endif()
        endforeach()
        if(NOT imported_location OR imported_location MATCHES "-NOTFOUND$")
            get_target_property(imported_location "${target}" IMPORTED_LOCATION_NOCONFIG)
        endif()
        if(NOT imported_implib OR imported_implib MATCHES "-NOTFOUND$")
            get_target_property(imported_implib "${target}" IMPORTED_IMPLIB_NOCONFIG)
        endif()
        if(imported_location MATCHES "-NOTFOUND$")
            set(imported_location "")
        endif()
        if(imported_implib MATCHES "-NOTFOUND$")
            set(imported_implib "")
        endif()
        string(TOLOWER "${imported_location}" imported_location_lower)
        if(imported_implib)
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "DYNAMIC|$<TARGET_LINKER_FILE:${target}>")
        elseif(target MATCHES "_static$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "STATIC|$<TARGET_LINKER_FILE:${target}>")
        elseif(imported_location_lower MATCHES "\\.dll\\.a$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "DYNAMIC|$<TARGET_LINKER_FILE:${target}>")
        elseif(imported_location_lower MATCHES "\\.a$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "STATIC|$<TARGET_LINKER_FILE:${target}>")
        elseif(target MATCHES "^(Vulkan::Vulkan|CUDA::)" OR
               imported_location_lower MATCHES "\\.(so(\\.[0-9.]+)?|dylib|dll|framework|tbd)$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "DYNAMIC|$<TARGET_LINKER_FILE:${target}>")
        elseif(imported_location_lower MATCHES "\\.lib$")
            message(FATAL_ERROR "Cannot distinguish static library from import library for imported target ${target}")
        else()
            message(FATAL_ERROR "Cannot classify imported native library target ${target}")
        endif()
    endif()

    get_target_property(link_options "${target}" INTERFACE_LINK_OPTIONS)
    if(link_options AND NOT link_options MATCHES "-NOTFOUND$")
        foreach(option IN LISTS link_options)
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "OPTION|${option}")
        endforeach()
    endif()

    set(collected_items "")
    get_target_property(link_items "${target}" LINK_LIBRARIES)
    get_target_property(interface_items "${target}" INTERFACE_LINK_LIBRARIES)
    if(link_items AND NOT link_items MATCHES "-NOTFOUND$")
        list(APPEND collected_items ${link_items})
    endif()
    if(target_type STREQUAL "INTERFACE_LIBRARY" OR imported)
        if(interface_items AND NOT interface_items MATCHES "-NOTFOUND$")
            list(APPEND collected_items ${interface_items})
        endif()
    endif()
    foreach(item IN LISTS collected_items)
        if(item MATCHES "^\\$<LINK_ONLY:(.*)>$")
            set(item "${CMAKE_MATCH_1}")
        elseif(item MATCHES "^\\$<BUILD_INTERFACE:(.*)>$")
            set(item "${CMAKE_MATCH_1}")
        endif()
        if(item MATCHES "^::@\\(" OR item STREQUAL "::@")
            continue()
        elseif(item STREQUAL "" OR item MATCHES "^(debug|optimized|general)$")
            continue()
        elseif(TARGET "${item}")
            gpq_collect_link_items("${item}")
        elseif(item MATCHES "::")
            message(FATAL_ERROR "Unresolved namespaced native link target in ${target}: ${item}")
        elseif(item MATCHES "^-framework$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "FRAMEWORK_NEXT|framework")
        elseif(item MATCHES "^-weak_framework$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "FRAMEWORK_NEXT|weak_framework")
        elseif(item MATCHES "^-Wl," OR item MATCHES "^-pthread$")
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "OPTION|${item}")
        elseif(item MATCHES "^\\$")
            message(FATAL_ERROR "Unsupported native link generator expression: ${item}")
        else()
            set_property(GLOBAL APPEND PROPERTY GPQ_LINK_ITEMS "SYSTEM|${item}")
        endif()
    endforeach()
    get_property(stack GLOBAL PROPERTY GPQ_LINK_STACK)
    list(REMOVE_ITEM stack "${target}")
    set_property(GLOBAL PROPERTY GPQ_LINK_STACK "${stack}")
endfunction()
